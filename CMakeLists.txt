cmake_minimum_required(VERSION 3.16)
project(iq4_extension_core VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_subdirectory(src/core)
add_subdirectory(src/display)
add_subdirectory(src/codec)
add_library(iq4_runtime_core STATIC src/runtime/recording.cpp)
target_include_directories(iq4_runtime_core PUBLIC src/runtime)
if(APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE IQ4_RUNTIME_SDK OUTPUT_STRIP_TRAILING_WHITESPACE)
  target_include_directories(iq4_runtime_core SYSTEM PUBLIC ${IQ4_RUNTIME_SDK}/usr/include/c++/v1)
endif()
add_executable(iq4_runtime_tests tests/runtime/test_recording.cpp)
target_link_libraries(iq4_runtime_tests PRIVATE iq4_runtime_core)
if(MSVC)
  target_compile_options(iq4_runtime_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(iq4_runtime_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME iq4_runtime_host_validation COMMAND iq4_runtime_tests)
add_subdirectory(src/recording)
option(IQ4_BUILD_SDK_DIAGNOSTICS "Compile SDK diagnostic code; never execute a camera test automatically" OFF)
if(IQ4_BUILD_SDK_DIAGNOSTICS)
  add_subdirectory(tools/sdk)
endif()
