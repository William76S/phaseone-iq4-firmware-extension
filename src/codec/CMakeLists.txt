cmake_minimum_required(VERSION 3.16)
project(iq4_bounded_codec LANGUAGES C CXX)
set(IQ4_CODEC_API_VERSION "80" CACHE STRING "Public header API: host stock 80 or unbound IQ4 layout 82")
if(NOT IQ4_CODEC_API_VERSION MATCHES "^(80|82)$")
  message(FATAL_ERROR "Only the checked JPEG8-family API layouts 80/82 are supported")
endif()
add_library(iq4_bounded_codec STATIC bounded_jpeg.c)
target_compile_features(iq4_bounded_codec PUBLIC c_std_11)
target_compile_definitions(iq4_bounded_codec PUBLIC IQ4_JPEG_API_VERSION=${IQ4_CODEC_API_VERSION})
target_include_directories(iq4_bounded_codec PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(iq4_bounded_codec PRIVATE -Wall -Wextra -Wpedantic)
# The adapter itself needs no linked libjpeg: every C entry point is injected.
set(IQ4_CODEC_HOST_LIBJPEG "" CACHE FILEPATH "Verified official 1.5.3 JPEG8 static host library")
option(IQ4_CODEC_BUILD_HOST_TESTS "Build real host JPEG encoding/decoding tests" OFF)
if(IQ4_CODEC_BUILD_HOST_TESTS)
  if(CMAKE_CROSSCOMPILING OR NOT IQ4_CODEC_API_VERSION STREQUAL "80")
    message(FATAL_ERROR "Host tests require API80 and a native host build")
  endif()
  if(NOT EXISTS "${IQ4_CODEC_HOST_LIBJPEG}")
    message(FATAL_ERROR "Set IQ4_CODEC_HOST_LIBJPEG to the locally verified JPEG8 library")
  endif()
  add_executable(iq4_codec_tests ../../tests/codec/test_bounded.cpp)
  target_compile_features(iq4_codec_tests PRIVATE cxx_std_17)
  target_link_libraries(iq4_codec_tests PRIVATE iq4_bounded_codec ${IQ4_CODEC_HOST_LIBJPEG})
  if(APPLE)
    execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE IQ4_CODEC_SDK OUTPUT_STRIP_TRAILING_WHITESPACE)
    target_include_directories(iq4_codec_tests SYSTEM PRIVATE ${IQ4_CODEC_SDK}/usr/include/c++/v1)
  endif()
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  enable_testing()
  add_test(NAME iq4_bounded_codec_host_validation
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/codec/run_host_validation.py
      --test-executable $<TARGET_FILE:iq4_codec_tests>
      --host-library ${IQ4_CODEC_HOST_LIBJPEG}
      --evidence-root ${CMAKE_CURRENT_SOURCE_DIR}/../../evidence/codec/runs)
endif()
