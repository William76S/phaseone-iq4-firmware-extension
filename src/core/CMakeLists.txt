cmake_minimum_required(VERSION 3.16)
project(iq4_image_core VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(iq4_image_core STATIC lib/image_core.cpp)
target_include_directories(iq4_image_core PUBLIC include)
if(APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE IQ4_HOST_SDK OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE IQ4_SDK_QUERY_RESULT)
  if(NOT IQ4_SDK_QUERY_RESULT EQUAL 0)
    message(FATAL_ERROR "Cannot obtain macOS SDK path from xcrun")
  endif()
  target_include_directories(iq4_image_core SYSTEM PUBLIC ${IQ4_HOST_SDK}/usr/include/c++/v1)
endif()
target_link_libraries(iq4_image_core PUBLIC Threads::Threads)
target_compile_options(iq4_image_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(iq4_core_tests ../../tests/core/test_core.cpp)
target_link_libraries(iq4_core_tests PRIVATE iq4_image_core)
target_compile_options(iq4_core_tests PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME iq4_core_host_validation COMMAND iq4_core_tests ${CMAKE_CURRENT_SOURCE_DIR}/../../fixtures/luts)
