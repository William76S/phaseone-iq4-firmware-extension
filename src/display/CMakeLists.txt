cmake_minimum_required(VERSION 3.16)
project(iq4_display_mask VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT TARGET iq4_image_core)
  add_subdirectory(../core ${CMAKE_CURRENT_BINARY_DIR}/core)
endif()
add_library(iq4_display_mask STATIC lib/display_mask.cpp)
target_include_directories(iq4_display_mask PUBLIC include)
target_link_libraries(iq4_display_mask PUBLIC iq4_image_core)
target_compile_options(iq4_display_mask PRIVATE -Wall -Wextra -Wpedantic)
add_executable(iq4_display_tests ../../tests/display/test_display.cpp)
target_link_libraries(iq4_display_tests PRIVATE iq4_display_mask)
target_compile_options(iq4_display_tests PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME iq4_display_host_validation COMMAND iq4_display_tests)
