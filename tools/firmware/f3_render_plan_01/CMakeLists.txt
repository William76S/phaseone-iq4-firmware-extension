cmake_minimum_required(VERSION 3.16)
project(iq4_f3_render_plan_01 LANGUAGES C CXX)
add_library(iq4_f3_render_plan_01 STATIC render_plan.c native_render_adapter.c core_receipt.c native_api_bridge.cpp)
target_include_directories(iq4_f3_render_plan_01 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(iq4_f3_render_plan_01 PUBLIC c_std_11)
target_compile_features(iq4_f3_render_plan_01 PRIVATE cxx_std_17)
option(IQ4_F3_RENDER_PLAN_TESTS "Build independent host contract tests" ON)
if(IQ4_F3_RENDER_PLAN_TESTS)
  enable_testing()
  foreach(unit test_plan test_native_adapter test_receipt test_api_bridge)
    add_executable(iq4_f3_${unit} ../../../tests/f3_render_plan_01/${unit}.cpp)
    target_compile_features(iq4_f3_${unit} PRIVATE cxx_std_17)
    target_link_libraries(iq4_f3_${unit} PRIVATE iq4_f3_render_plan_01)
    add_test(NAME iq4_f3_${unit} COMMAND iq4_f3_${unit})
  endforeach()
endif()
