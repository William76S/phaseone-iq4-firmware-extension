cmake_minimum_required(VERSION 3.16)
project(iq4_f4_owned_copy_pool LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(iq4_f4_owned_copy STATIC owned_copy_pool.cpp recorder_worker.cpp)
target_include_directories(iq4_f4_owned_copy PUBLIC .)
target_link_libraries(iq4_f4_owned_copy PUBLIC Threads::Threads)
if(APPLE)
  execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE IQ4_POOL_SDK OUTPUT_STRIP_TRAILING_WHITESPACE)
  target_include_directories(iq4_f4_owned_copy SYSTEM PUBLIC ${IQ4_POOL_SDK}/usr/include/c++/v1)
endif()
if(MSVC)
  target_compile_options(iq4_f4_owned_copy PRIVATE /W4 /WX)
else()
  target_compile_options(iq4_f4_owned_copy PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
option(IQ4_POOL_SANITIZE "Host ASan+UBSan tests" OFF)
option(IQ4_POOL_THREAD_SANITIZE "Host thread sanitizer tests" OFF)
if(IQ4_POOL_SANITIZE AND IQ4_POOL_THREAD_SANITIZE)
  message(FATAL_ERROR "Choose ASan/UBSan or TSan separately")
endif()
if(IQ4_POOL_SANITIZE OR IQ4_POOL_THREAD_SANITIZE)
  if(IQ4_POOL_SANITIZE)
    set(IQ4_POOL_SANITIZER -fsanitize=address,undefined)
  else()
    set(IQ4_POOL_SANITIZER -fsanitize=thread)
  endif()
  target_compile_options(iq4_f4_owned_copy PUBLIC ${IQ4_POOL_SANITIZER} -fno-omit-frame-pointer)
  target_link_options(iq4_f4_owned_copy PUBLIC ${IQ4_POOL_SANITIZER})
endif()
add_executable(iq4_f4_owned_copy_tests test_owned_copy_pool.cpp ../../../src/runtime/recording.cpp)
target_link_libraries(iq4_f4_owned_copy_tests PRIVATE iq4_f4_owned_copy)
if(MSVC)
  target_compile_options(iq4_f4_owned_copy_tests PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(iq4_f4_owned_copy_tests PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
endif()
enable_testing()
add_test(NAME iq4_f4_owned_copy_host_validation COMMAND iq4_f4_owned_copy_tests)
